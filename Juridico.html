/**
 * Controle Jurídico – Apps Script
 * 1) Envia e-mails (igual à versão anterior)
 * 2) Guarda e devolve os dados da página (registros, observações, usuários, e-mails)
 *    num arquivo JSON no seu Google Drive: "controle-juridico-dados.json"
 */
var NOME_ARQUIVO = 'controle-juridico-dados.json';

// Rode esta função UMA vez no editor (botão ▶) para autorizar Drive e e-mail.
function autorizar() {
  getArquivo_();
  MailApp.getRemainingDailyQuota();
}

function json_(o) {
  return ContentService.createTextOutput(JSON.stringify(o))
    .setMimeType(ContentService.MimeType.JSON);
}

function getArquivo_() {
  var props = PropertiesService.getScriptProperties();
  var id = props.getProperty('ARQUIVO_ID');
  if (id) {
    try { return DriveApp.getFileById(id); } catch (e) {}
  }
  var f = DriveApp.createFile(NOME_ARQUIVO, JSON.stringify({ rev: 0, data: null }), MimeType.PLAIN_TEXT);
  props.setProperty('ARQUIVO_ID', f.getId());
  return f;
}

function ler_() {
  var txt = getArquivo_().getBlob().getDataAsString();
  return txt ? JSON.parse(txt) : { rev: 0, data: null };
}

// Página pergunta: "tem novidade?" (?rev=N) ou carrega tudo.
function doGet(e) {
  var rev = Number(PropertiesService.getScriptProperties().getProperty('REV') || 0);
  var pedido = e && e.parameter && e.parameter.rev;
  if (rev > 0 && pedido !== undefined && Number(pedido) === rev) {
    return json_({ ok: true, rev: rev, unchanged: true });
  }
  var s = ler_();
  return json_({ ok: true, rev: s.rev, data: s.data });
}

function doPost(e) {
  var d = JSON.parse(e.postData.contents);

  // Salvar dados da página
  if (d.action === 'save') {
    var lock = LockService.getScriptLock();
    lock.waitLock(20000);
    try {
      var atual = ler_();
      if (Number(d.baseRev) !== atual.rev) {
        // alguém salvou antes: devolve a versão atual para a página mesclar
        return json_({ ok: false, conflict: true, rev: atual.rev, data: atual.data });
      }
      var novo = { rev: atual.rev + 1, data: d.data, atualizado: new Date().toISOString() };
      getArquivo_().setContent(JSON.stringify(novo));
      PropertiesService.getScriptProperties().setProperty('REV', String(novo.rev));
      return json_({ ok: true, rev: novo.rev });
    } finally {
      lock.releaseLock();
    }
  }

  // Padrão: enviar e-mail
  MailApp.sendEmail({ to: d.to, subject: d.subject, body: d.body, htmlBody: d.html });
  return json_({ ok: true });
}
